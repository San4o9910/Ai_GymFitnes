function render(){
  document.querySelectorAll('.nav-item').forEach(b=>b.classList.toggle('active',b.dataset.route===state.route));
  if(state.route==='today') renderToday();
  else if(state.route==='schedule') renderSchedule();
  else if(state.route==='progress') renderProgress();
  else if(state.route==='library') renderLibrary();
  else renderSettings();
  bindCommon();
}
