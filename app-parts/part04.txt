function workoutProgress(workoutId,light){
  const w=workouts[workoutId]; const exs=light?w.exercises.slice(0,5):w.exercises; let total=0,done=0;
  exs.forEach(([id,sets])=>{ const actual=light&&id!=='warmup'?Math.min(2,sets):sets; total+=actual; const key=`${todayISO()}_${workoutId}_${id}`; done+=(state.setLogs[key]||[]).filter(x=>x.done).length; });
  return total?Math.min(100,Math.round(done/total*100)):0;
}
